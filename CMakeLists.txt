cmake_minimum_required(VERSION 3.16)
project(KirbyCouchCoop C CXX)
set(BUILD_LIBRETRO ON CACHE BOOL "" FORCE)
set(SKIP_LIBRARY ON CACHE BOOL "" FORCE)
set(DISABLE_DEPS ON CACHE BOOL "" FORCE)
set(DISABLE_FRONTENDS ON CACHE BOOL "" FORCE)
set(BUILD_GL OFF CACHE BOOL "" FORCE)
set(BUILD_GLES2 OFF CACHE BOOL "" FORCE)
set(BUILD_GLES3 OFF CACHE BOOL "" FORCE)
set(BUILD_LTO OFF CACHE BOOL "" FORCE)
set(ENABLE_DEBUGGERS OFF CACHE BOOL "" FORCE)
set(ENABLE_SCRIPTING OFF CACHE BOOL "" FORCE)
set(M_CORE_GB OFF CACHE BOOL "" FORCE)
add_subdirectory(mgba)
get_target_property(MGBA_RETRO_SOURCES mgba_libretro SOURCES)
list(FILTER MGBA_RETRO_SOURCES EXCLUDE REGEX "/src/platform/libretro/libretro[.]c$")
list(FILTER MGBA_RETRO_SOURCES EXCLUDE REGEX "/src/platform/windows/vfs-w32[.]c$")
set_property(TARGET mgba_libretro PROPERTY SOURCES "${MGBA_RETRO_SOURCES}")
target_sources(mgba_libretro PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/coop.c")
target_sources(mgba_libretro PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mgba/src/gba/sio/lockstep.c")
target_compile_definitions(mgba_libretro PRIVATE ENABLE_VFS_FD)
target_include_directories(mgba_libretro PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mgba/src/platform/libretro")
set_target_properties(mgba_libretro PROPERTIES OUTPUT_NAME kirby_coop_libretro)
if(ANDROID)
    target_link_options(mgba_libretro PRIVATE "-Wl,-z,max-page-size=16384")
endif()
